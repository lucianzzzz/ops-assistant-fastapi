### 1.2 模块划分原则

- **按业务领域划分**，不是按技术类型
- **单一职责**：每个模块只负责一个领域
- **依赖方向**：`api → core → repository`，禁止反向依赖
- **通用代码**：放在 `core/common/`

## 三、异常处理

### 3.1 自定义异常体系

```python
# app/core/common/exceptions.py

class OpsAssistantError(Exception):
    """基础异常类"""
    def __init__(self, message: str, code: str = "INTERNAL_ERROR"):
        self.message = message
        self.code = code
        super().__init__(message)

# 业务异常
class KnowledgeNotFoundError(OpsAssistantError):
    """知识库未找到"""
    def __init__(self, question: str):
        super().__init__(
            message=f"未找到问题 '{question}' 的相关知识",
            code="KNOWLEDGE_NOT_FOUND"
        )

class InvalidProvinceError(OpsAssistantError):
    """无效的省份参数"""
    pass

# 外部依赖异常
class AIServiceError(OpsAssistantError):
    """AI服务异常"""
    pass

class DatabaseError(OpsAssistantError):
    """数据库异常"""
    pass
```

### 3.2 异常处理原则

```python
# 1. 具体异常优先捕获
try:
    result = search_knowledge(question)
except KnowledgeNotFoundError as e:
    logger.warning(f"知识未找到: {e.message}")
    return fallback_result
except DatabaseError as e:
    logger.error(f"数据库错误: {e.message}", exc_info=True)
    raise HTTPException(status_code=500, detail="服务暂时不可用")

# 2. 记录异常上下文
try:
    ai_result = call_ai_service(question)
except Exception as e:
    logger.error(
        "AI服务调用失败",
        extra={
            "question": question,
            "error": str(e),
            "error_type": type(e).__name__
        },
        exc_info=True
    )
    raise AIServiceError("AI服务暂时不可用") from e

# 3. 禁止裸 except
# ❌ 错误
try:
    do_something()
except:  # 裸except
    pass

# ✅ 正确
try:
    do_something()
except Exception as e:  # 明确捕获Exception
    logger.error(f"操作失败: {e}")
    raise
```

### 3.3 API层异常转换

```python
# app/middleware/error_handler.py

from fastapi import Request, status
from fastapi.responses import JSONResponse
from app.core.common.exceptions import OpsAssistantError

@app.exception_handler(OpsAssistantError)
async def ops_error_handler(request: Request, exc: OpsAssistantError):
    """统一处理业务异常"""
    return JSONResponse(
        status_code=status.HTTP_400_BAD_REQUEST,
        content={
            "success": False,
            "error": {
                "code": exc.code,
                "message": exc.message
            }
        }
    )

@app.exception_handler(Exception)
async def global_error_handler(request: Request, exc: Exception):
    """捕获未处理的异常"""
    logger.error(f"未处理异常: {exc}", exc_info=True)
    return JSONResponse(
        status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
        content={
            "success": False,
            "error": {
                "code": "INTERNAL_ERROR",
                "message": "服务内部错误"
            }
        }
    )
```

### 4.1 日志配置

```python
# app/core/common/logging.py

import logging
import sys
from pathlib import Path

def setup_logging(log_level: str = "INFO", log_file: Path | None = None):
    """配置日志系统"""
    
    # 格式化
    formatter = logging.Formatter(
        fmt="%(asctime)s | %(levelname)-8s | %(name)s:%(lineno)d | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S"
    )
    
    # 控制台输出
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setFormatter(formatter)
    
    handlers = [console_handler]
    
    # 文件输出（可选）
    if log_file:
        file_handler = logging.FileHandler(log_file, encoding="utf-8")
        file_handler.setFormatter(formatter)
        handlers.append(file_handler)
    
    # 根日志配置
    logging.basicConfig(
        level=getattr(logging, log_level.upper()),
        handlers=handlers
    )
    
    # 关闭第三方库的DEBUG日志
    logging.getLogger("httpx").setLevel(logging.WARNING)
    logging.getLogger("chromadb").setLevel(logging.WARNING)
```

### 5.2 统一响应格式

```python
# app/core/models/base.py

from typing import Generic, Optional, TypeVar
from pydantic import BaseModel

T = TypeVar("T")

class ApiResponse(BaseModel, Generic[T]):
    """统一API响应"""
    success: bool
    data: Optional[T] = None
    error: Optional[dict] = None
    
    @classmethod
    def ok(cls, data: T) -> "ApiResponse[T]":
        """成功响应"""
        return cls(success=True, data=data)
    
    @classmethod
    def fail(cls, code: str, message: str) -> "ApiResponse":
        """失败响应"""
        return cls(
            success=False,
            error={"code": code, "message": message}
        )

# 使用示例
@router.post("/ask", response_model=ApiResponse[AskResponse])
async def ask_question(request: AskRequest):
    try:
        result = service.search(request.question)
        return ApiResponse.ok(data=result)
    except KnowledgeNotFoundError as e:
        return ApiResponse.fail(code=e.code, message=e.message)
```

### 5.3 请求参数验证

```python
from pydantic import BaseModel, Field, validator

class AskRequest(BaseModel):
    """问答请求"""
    question: str = Field(..., min_length=1, max_length=500, description="用户问题")
    province: Optional[str] = Field(None, description="省份过滤")
    top_k: int = Field(3, ge=1, le=10, description="返回结果数量")
    
    @validator("question")
    def question_not_empty(cls, v: str) -> str:
        """验证问题非空"""
        if not v.strip():
            raise ValueError("问题不能为空")
        return v.strip()
    
    @validator("province")
    def province_valid(cls, v: Optional[str]) -> Optional[str]:
        """验证省份有效性"""
        if v is None:
            return v
        valid_provinces = ["浙江", "江苏", "广东", "四川", "北京"]
        if v not in valid_provinces:
            raise ValueError(f"省份必须是: {', '.join(valid_provinces)}")
        return v
```

### 7.1 服务依赖

```python
# app/core/common/dependencies.py

from functools import lru_cache
from typing import Annotated
from fastapi import Depends

from app.core.assistant.repository import KnowledgeRepository
from app.core.assistant.service import AssistantService

@lru_cache()
def get_repository() -> KnowledgeRepository:
    """获取数据仓库单例"""
    return KnowledgeRepository()

def get_assistant_service(
    repository: Annotated[KnowledgeRepository, Depends(get_repository)]
) -> AssistantService:
    """获取问答服务"""
    return AssistantService(repository=repository)

# 使用
@router.post("/ask")
async def ask(
    request: AskRequest,
    service: Annotated[AssistantService, Depends(get_assistant_service)]
):
    return service.search(request.question)
```
